5//! This provides emulator portion of the toolbar.
6//!
7
8use eframe::epaint::Color32;
9use egui::{Button, RichText, Ui};
10use crate::emulator::emu_debug::CtrlMSG;
11use crate::TitoApp;
12
13impl TitoApp {
14    pub fn emulator_toolbar(&mut self, ui: &mut Ui) {
15        // Power button
16        let text_onoff = match self.emu_running {
17            true => RichText::new("⏼on/off").color(Color32::WHITE),
18            false => RichText::new("⏼on/off"),
19        };
20        if ui.selectable_label(self.emu_running, text_onoff).clicked() {
21            self.emu_playing = false;
22            let _ = self.tx_ctrl.send(CtrlMSG::PlaybackPlayPause(false));
23            if self.emu_running {
24                self.stop_emulation();
25            } else {
26                self.emu_running = true;
27                let _ = self.tx_ctrl.send(CtrlMSG::EnableBreakpoints(self.config.memview_breakpoints_enabled));
28                let _ = self.tx_ctrl.send(CtrlMSG::PlaybackStart);
29            }
30        }
31
32        ui.add_enabled_ui(self.emu_running, |ui| {
33            // PlayPause Button
34            let text_play;
35            match self.emu_playing {
36                true => text_play = RichText::new("⏸").color(Color32::WHITE),
37                false => text_play = RichText::new("▶"),
38            }
39            if ui.add(Button::new(text_play).min_size(egui::vec2(24.0, 0.0))).clicked() {
40                self.emu_playing = !self.emu_playing;
41                let _ = self.tx_ctrl
42                    .send(CtrlMSG::PlaybackPlayPause(self.emu_playing));
43            }
44            // Step Button
45            ui.add_enabled_ui(!self.emu_playing, |ui| {
46                if ui.add(Button::new(RichText::new("|▶")).min_size(egui::vec2(24.0, 0.0))).clicked() {
47                    let _ = self.tx_ctrl.send(CtrlMSG::PlaybackTick);
48                    if self.config.memview_follow_pc {
49                        self.memoryview.jump_to_pc();
50                    }
51                }
52            })
53        });
54
55        ui.separator();
56
57        if ui.button("Reset").clicked() {
58            let _ = self.tx_ctrl.send(CtrlMSG::Reset());
59            self.legacytermview.clear();
60            self.graphicsview.clear();
61        }
62        ui.separator();
63    }
64}
