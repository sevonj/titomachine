5//! This module contains the Legacy Terminal Panel: =CRT and =KBD
6//!
7
8use std::sync::mpsc::{Receiver, Sender};
9use egui::{Color32, Frame, RichText, Ui, TextEdit, Stroke, TopBottomPanel, Button};
10use crate::config::Config;
11use crate::emulator::emu_debug::CtrlMSG;
12use crate::gui::EmulatorPanel;
13
14// The space at the end prevents buf_crt.lines() from dropping the last line.
15const CRT_CLEAR_TEXT: &str = "\n\n\n\n\n\n\n\n\n\n\n\n\n\n\n ";
16const COLOR_INPUTHIGHLIGHT: Color32 = Color32::YELLOW;
17
18/// LegacyTermView is the UI component responsible for the memory viewer panel.
19pub(crate) struct LegacyTermView {
20    rx_crt: Receiver<i32>,
21    tx_kbd: Sender<i32>,
22    rx_kbdreq: Receiver<()>,
23    buf_kbd: String,
24    buf_crt: String,
25
26    waiting_for_input: bool,
27}
28
29impl LegacyTermView {
30    pub fn new(rx_crt: Receiver<i32>, tx_kbd: Sender<i32>, rx_kbdreq: Receiver<()>) -> Self {
31        LegacyTermView {
32            rx_crt,
33            tx_kbd,
34            rx_kbdreq,
35            buf_kbd: String::new(),
36            buf_crt: CRT_CLEAR_TEXT.to_owned(),
37            waiting_for_input: false,
38        }
39    }
40
41    /// Add a value to CRT buffer
42    pub fn crt_out(&mut self, n: i32) {
43        // Remove top line
44        self.buf_crt = self.buf_crt.lines().skip(1).map(|s| s.to_string() + "\n").collect();
45        // Insert line to the end
46        self.buf_crt += n.to_string().as_str();
47    }
48
49    pub fn clear(&mut self) {
50        self.buf_kbd = String::new();
51        self.buf_crt = CRT_CLEAR_TEXT.to_owned();
52        self.unjam_input_wait();
53    }
54
55    /// If the emulator thread is waiting for input, it's frozen until it receives something.
56    /// This will free the emulator by sending a dummy value.
57    pub fn unjam_input_wait(&mut self) {
58        if !self.waiting_for_input {
59            return;
60        }
61        let _ = self.tx_kbd.send(0);
62    }
63}
64
65impl EmulatorPanel for LegacyTermView {
66    fn ui(&mut self, ui: &mut Ui, config: &mut Config, _sender: &Sender<CtrlMSG>) {
67        // Update
68        if let Ok(n) = self.rx_crt.try_recv() {
69            self.crt_out(n)
70        }
71        if let Ok(_) = self.rx_kbdreq.try_recv() {
72            self.waiting_for_input = true;
73            // Pop the panel open if it's needed!
74            config.legacyterm_visible = true;
75        }
76
77        // LegacyTerm titlebar
78        TopBottomPanel::top("legacyterm_titlebar")
79            .resizable(false)
80            .show_inside(ui, |ui| {
81                ui.horizontal(|ui| {
82                    let toggle_text = if config.legacyterm_visible { "⏷ Legacy Terminal" } else { "⏵ Legacy Terminal" };
83                    if ui.add(Button::new(toggle_text).frame(false)).clicked() {
84                        config.legacyterm_visible = !config.legacyterm_visible;
85                    }
86                    if !config.legacyterm_visible {
87                        return;
88                    }
89                });
90            });
91
92        if !config.legacyterm_visible {
93            return;
94        }
95
96        TopBottomPanel::top("legacyterm_main")
97            .resizable(false)
98            .show_inside(ui, |ui| {
99                // =CRT
100                ui.label("=CRT");
101                Frame::none()
102                    .fill(Color32::BLACK)
103                    .show(ui, |ui| {
104                        ui.label(self.buf_crt.as_str());
105                        ui.allocate_space(egui::vec2(ui.available_width(), 0.0))
106                    });
107                ui.separator();
108
109                // =KBD
110                ui.add_enabled_ui(self.waiting_for_input, |ui| {
111                    // KBD Label
112                    if self.waiting_for_input {
113                        Frame::none()
114                            .stroke(Stroke { width: 1.0, color: COLOR_INPUTHIGHLIGHT })
115                            .show(ui, |ui| {
116                                ui.label(
117                                    RichText::new("=KBD")
118                                        .strong()
119                                        .color(Color32::WHITE),
120                                );
121                            });
122                    } else {
123                        ui.label(
124                            RichText::new("=KBD")
125                                .strong()
126                                .color(Color32::WHITE),
127                        );
128                    }
129                    ui.horizontal(|ui| {
130                        ui.spacing_mut().item_spacing.x = 4.0;
131                        // KBD text field
132                        TextEdit::singleline(&mut self.buf_kbd).hint_text("Type a value").desired_width(78.0).show(ui);
133                        // KBD send button
134                        if ui.button("⬈").clicked() {
135                            if self.buf_kbd.parse::<i32>().is_ok() {
136                                let _ = self.tx_kbd.send(self.buf_kbd.parse::<i32>().unwrap());
137                                self.buf_kbd = String::new();
138                                self.waiting_for_input = false;
139                            } else {
140                                self.buf_kbd = "Invalid input!".to_owned();
141                            }
142                        }
143                    });
144                    ui.add_space(4.0);
145                });
146            });
147    }
148}
