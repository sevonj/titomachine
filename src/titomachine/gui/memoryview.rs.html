22const MEM_SIZE: usize = 0x2000;
23const COLOR_SEGMENT_NONE: Color32 = Color32::from_rgb(60, 60, 60);
24const COLOR_SEGMENT_CODE: Color32 = Color32::from_rgb(167, 115, 0);
25const COLOR_SEGMENT_DATA: Color32 = Color32::from_rgb(046, 137, 133);
26const COLOR_SEGMENT_STACK: Color32 = Color32::from_rgb(159, 075, 150);
27const COLOR_BREAKPOINT: Color32 = Color32::from_rgb(239, 80, 57);
28const COLOR_BREAKPOINT_OPTION: Color32 = Color32::from_rgb(228, 122, 119);
29const COLOR_BREAKPOINT_DISABLED: Color32 = Color32::from_additive_luminance(0x1f);
30
31/// Which segment does an address belong to.
32enum MemorySegment {
33    None,
34    Code,
35    Data,
36    Stack,
37}
38
39/// MemoryView is the UI component responsible for the memory viewer panel.
40pub(crate) struct MemoryView {
41    /// First address to cache
42    view_cache_start: usize,
43    /// Number of addresses to cache
44    view_cache_size: usize,
45    /// Currently cached addresses
46    view_cache: HashMap<usize, i32>,
47
48    /// Multiple symbols may exist for an address, because of _consts_
49    symbol_table: HashMap<usize, Vec<String>>,
50    /// Set of addresses that contains a breakpoint.
51    breakpoints: HashSet<usize>,
52
53    /// Is the emulated machine both turned on and not paused
54    pub is_playing: bool,
55
56    /// Memory view displays where PC, SP, and FP point to
57    pub cpu_pc: usize,
58    /// Memory view displays where PC, SP, and FP point to
59    pub cpu_sp: usize,
60    /// Memory view displays where PC, SP, and FP point to
61    pub cpu_fp: usize,
62
63    /// Code segment start address
64    pub start_code: usize,
65    /// Data segment start address
66    pub start_data: usize,
67    /// Stack start address
68    pub start_stack: usize,
69}
70
71impl MemoryView {
72    pub fn new() -> Self {
73        MemoryView {
74            view_cache_start: 0,
75            view_cache_size: 32,
76            view_cache: HashMap::new(),
77            symbol_table: HashMap::new(),
78            breakpoints: HashSet::new(),
79            is_playing: false,
80            cpu_pc: 0,
81            cpu_sp: 0,
82            cpu_fp: 0,
83            start_code: MEM_SIZE,
84            start_data: MEM_SIZE,
85            start_stack: MEM_SIZE,
86        }
87    }
88
89    /// Reset needs to be called when loading a new program. Otherwise, old stuff like breakpoints
90    /// may linger in the memory view despite being removed from the emulator.
91    pub fn reset(&mut self) {
92        self.view_cache_start = 0;
93        self.symbol_table.clear();
94        self.breakpoints.clear();
95        self.start_code = MEM_SIZE;
96        self.start_data = MEM_SIZE;
97        self.start_stack = MEM_SIZE;
98    }
99
100    /// Range of addresses currently visible on screen
101    pub fn get_view_cache_range(&self) -> Range<u32> {
102        (self.view_cache_start as u32)..(self.view_cache_start + self.view_cache_size + 1) as u32
103    }
104    /// First address currently visible on screen
105    pub fn get_view_cache_start(&self) -> usize {
106        self.view_cache_start
107    }
108
109    /// Update memoryview's cached addresses. range_start tells the first address.
110    /// Use get_view_cache_range() to know which addresses memoryview wants.
111    pub fn set_view_cache(&mut self, range_start: usize, values: Vec<i32>) {
112        //self.view_cache.clear();
113        for (offset, value) in values.iter().enumerate() {
114            let address = range_start + offset;
115            self.view_cache.insert(address, value.to_owned());
116        }
117    }
118
119    /// Give memoryview a copy of B91 symbol table.
120    pub fn set_symbol_table(&mut self, table: HashMap<String, i32>) {
121        self.symbol_table.clear();
122        for (name, value) in table {
123            // Negative addresses don't exist, they wouldn't show up anyway.
124            if value < 0 {
125                continue;
126            }
127            let addr = value as usize;
128            match self.symbol_table.get_mut(&addr) {
129                None => {
130                    self.symbol_table.insert(addr, vec![name]);
131                }
132                Some(vec) => {
133                    vec.push(name);
134                }
135            }
136        }
137    }
138
139    /// Scroll view to PC location
140    pub fn jump_to_pc(&mut self) {
141        if self.cpu_pc > 4 {
142            self.view_cache_start = self.cpu_pc - 4;
143        } else {
144            self.view_cache_start = 0
145        }
146    }
147
148    /// Which segment does an address belong?
149    fn get_segment_from_address(&self, address: usize) -> MemorySegment {
150        if address >= MEM_SIZE {
151            MemorySegment::None
152        } else if address >= self.start_stack {
153            MemorySegment::Stack
154        } else if address >= self.start_data {
155            MemorySegment::Data
156        } else if address >= self.start_code {
157            MemorySegment::Code
158        } else {
159            MemorySegment::None
160        }
161    }
162
163    /// Determine how many table rows can fit on screen. May give one too many, but overflow should
164    /// be hidden.
165    fn get_table_rows_fit(&self, ui: &Ui) -> usize {
166        let row_height = 23.0;
167        (ui.available_height() / row_height) as usize
168    }
169
170    /// Add an address-entry-displaying row to the memory view table.
171    fn add_table_row(
172        &mut self,
173        config: &mut Config,
174        sender: &Sender<CtrlMSG>,
175        body: &mut TableBody,
176        address: usize,
177    ) {
178        // Out of bounds
179        if address >= MEM_SIZE {
180            return;
181        }
182        let font_color = match self.cpu_pc == address {
183            true => COL_TEXT_HI,
184            false => COL_TEXT
185        };
186
187        match self.view_cache.get(&address) {
188
189            // Display entry
190            Some(value) => {
191                let value = value.to_owned();
192                body.row(20.0, |mut row| {
193                    self.add_table_address(config, sender, &mut row, address, font_color);
194                    self.add_table_value(config, &mut row, value, font_color);
195                    self.add_table_disassembly(&mut row, value, font_color);
196                    self.add_table_pointers(&mut row, address, font_color);
197                });
198            }
199
200            // Display placeholder
201            None => {
202                body.row(20.0, |mut row| {
203                    self.add_table_address(config, sender, &mut row, address, font_color);
204                    self.add_table_label(&mut row, "Fetching...", font_color);
205                    self.add_table_label(&mut row, "", font_color);
206                    self.add_table_label(&mut row, "", font_color);
207                });
208            }
209        }
210    }
211
212    /// Table Shortcut: Generic label
213    fn add_table_label(&self, row: &mut TableRow, text: &str, font_color: Color32) {
214        row.col(|ui| {
215            ui.label(RichText::new(text).font(FONT_TBL.clone()).color(font_color));
216        });
217    }
218
219    /// Table Shortcut: Address column
220    fn add_table_address(
221        &mut self,
222        config: &Config,
223        sender: &Sender<CtrlMSG>,
224        row: &mut TableRow,
225        address: usize,
226        font_color: Color32) {
227        let text = config.memview_addr_base.format_addr(address);
228        let color = match self.get_segment_from_address(address) {
229            MemorySegment::None => COLOR_SEGMENT_NONE,
230            MemorySegment::Code => COLOR_SEGMENT_CODE,
231            MemorySegment::Data => COLOR_SEGMENT_DATA,
232            MemorySegment::Stack => COLOR_SEGMENT_STACK
233        };
234        row.col(|ui| {
235            // Segment marker
236            ui.add(Image::new(include_image!("../assets/memview_segment_marker.png"))
237                .fit_to_original_size(1.0).tint(color)
238            );
239
240            // Address label
241            let addr_label = ui.label(RichText::new(text)
242                .font(FONT_TBL.clone())
243                .color(font_color)
244            );
245
246            // Breakpoints
247            let bp_color = if self.breakpoints.contains(&address) {
248                match config.memview_breakpoints_enabled {
249                    true => COLOR_BREAKPOINT,
250                    false => COLOR_BREAKPOINT_DISABLED
251                }
252            } else if addr_label.hovered() {
253                COLOR_BREAKPOINT_OPTION
254            } else {
255                Color32::TRANSPARENT
256            };
257
258            let bpmark = ui.add(Image::new(include_image!("../assets/memview_breakpoint.png"))
259                .fit_to_original_size(1.0).tint(bp_color)
260                .sense(Sense { click: true, drag: false, focusable: false })
261            );
262
263            match self.breakpoints.contains(&address) {
264                false => if addr_label.clicked() {
265                    self.breakpoints.insert(address);
266                    sender.send(CtrlMSG::InsertBreakpoint(address)).unwrap()
267                }
268                true => if bpmark.clicked() || addr_label.clicked() {
269                    self.breakpoints.remove(&address);
270                    sender.send(CtrlMSG::RemoveBreakpoint(address)).unwrap()
271                }
272            }
273        });
274    }
275
276    /// Table Shortcut: Value column
277    fn add_table_value(&self, config: &mut Config, row: &mut TableRow, value: i32, font_color: Color32) {
278        let text = config.memview_value_base.format_i32(value.to_owned());
279        row.col(|ui| {
280            ui.label(RichText::new(text).font(FONT_TBL.clone()).color(font_color));
281        });
282    }
283
284    /// Table Shortcut: Disassembly column
285    fn add_table_disassembly(&self, row: &mut TableRow, value: i32, font_color: Color32) {
286        let binding = disassemble_instruction(value.clone());
287        let text = binding.as_str();
288        row.col(|ui| {
289            ui.label(RichText::new(text).font(FONT_TBL.clone()).color(font_color));
290        });
291    }
292
293    /// Add a column that shows if PC, SP, FP, or any symbols point to this address
294    fn add_table_pointers(&self, row: &mut TableRow, address: usize, font_color: Color32) {
295        let mut text = String::new();
296        let symbols = self.symbol_table.get(&address);
297
298        if self.cpu_pc == address { text += "PC "; }
299        if self.cpu_sp == address { text += "SP "; }
300        if self.cpu_fp == address { text += "FP "; }
301
302        if let Some(vec) = symbols {
303            for symbol in vec {
304                text += format!("{} ", symbol).as_str()
305            }
306        }
307
308        row.col(|ui| {
309            if self.cpu_pc == address || self.cpu_sp == address || self.cpu_fp == address || symbols.is_some() {
310                // TODO: Make overlap
311                ui.add(Image::new(include_image!("../assets/memview_pointer_arrow.png"))
312                    .fit_to_original_size(1.0)
313                    .tint(Color32::from_rgba_unmultiplied(255, 255, 255, 2))
314                );
315            }
316            ui.label(RichText::new(text).font(FONT_TBL.clone()).color(font_color));
317        });
318    }
319
320    /// Table Shortcut: Header column
321    fn add_table_heading(&self, header: &mut TableRow, title: &str) {
322        header.col(|ui| {
323            ui.heading(RichText::new(title).font(FONT_TBLH.clone()));
324        });
325    }
326}

328impl EmulatorPanel for MemoryView {
329    fn ui(&mut self, ui: &mut Ui, config: &mut Config, sender: &Sender<CtrlMSG>) {
330        // Memview titlebar
331        TopBottomPanel::top("memview_titlebar")
332            .resizable(false)
333            .show_inside(ui, |ui| {
334                ui.horizontal(|ui| {
335                    let toggle_text = if config.memview_visible { "⏷ Memory Explorer" } else { "⏵ Memory Explorer" };
336                    if ui.add(Button::new(toggle_text).frame(false)).clicked(){
337                        config.memview_visible = !config.memview_visible;
338                    }
339                    if !config.memview_visible {
340                        return;
341                    }
342                    ui.menu_button("Options", |ui| {
343                        if ui.checkbox(&mut config.memview_follow_pc, "Follow PC").clicked() { ui.close_menu(); };
344                        ui.label("Display address as");
345                        if ui.radio_value(&mut config.memview_addr_base, Radix::Bin, "Binary").clicked() { ui.close_menu(); };
346                        if ui.radio_value(&mut config.memview_addr_base, Radix::Dec, "Decimal").clicked() { ui.close_menu(); };
347                        if ui.radio_value(&mut config.memview_addr_base, Radix::Hex, "Hex").clicked() { ui.close_menu(); };
348                        ui.label("Display value as");
349                        if ui.radio_value(&mut config.memview_value_base, Radix::Bin, "Binary").clicked() { ui.close_menu(); };
350                        if ui.radio_value(&mut config.memview_value_base, Radix::Dec, "Decimal").clicked() { ui.close_menu(); };
351                        if ui.radio_value(&mut config.memview_value_base, Radix::Hex, "Hex").clicked() { ui.close_menu(); };
352                        ui.label("Breakpoints");
353                        if ui.checkbox(&mut config.memview_breakpoints_enabled, "Enabled").clicked() {
354                            let _ = sender.send(CtrlMSG::EnableBreakpoints(config.memview_breakpoints_enabled));
355                            ui.close_menu();
356                        }
357                        if ui.button("Clear all").clicked() {
358                            self.breakpoints.clear();
359                            let _ = sender.send(CtrlMSG::ClearBreakpoints);
360                            ui.close_menu();
361                        }
362                    });
363                    if ui.button("Go to PC").clicked() {
364                        self.jump_to_pc();
365                    }
366                });
367            });
368
369        if !config.memview_visible {
370            return;
371        }
372
373        // Memview scrollbar
374        SidePanel::right("memview_scroll")
375            .resizable(false)
376            .max_width(22.0)
377            .frame(Frame::none())
378            .show_separator_line(false)
379            .show_inside(ui, |ui| {
380                // Space accounts for table header row
381                ui.add_space(20.);
382                ui.spacing_mut().slider_width = ui.available_height();
383                ui.add(Slider::new(&mut self.view_cache_start, 0x1fff..=0)
384                    .vertical()
385                    .smart_aim(false)
386                    .show_value(false)
387                    .handle_shape(egui::style::HandleShape::Rect { aspect_ratio: 1.6 })
388                );
389            });
390
391        // Memview main panel
392        CentralPanel::default()
393            .show_inside(ui, |ui| {
394                let rows_to_display = self.get_table_rows_fit(ui);
395                self.view_cache_size = rows_to_display;
396
397                // Mouse scroll
398                let scroll = -ui.input(|i| i.raw_scroll_delta).y.to_isize().unwrap().clamp(-1, 1);
399                self.view_cache_start = self.view_cache_start.saturating_add_signed(scroll);
400
401                // Follow PC
402                if config.memview_follow_pc & &self.is_playing {
403                    self.jump_to_pc();
404                }
405
406                // The purpose of this non-interactive ScrollArea is to hide overflow.
407                ScrollArea::vertical()
408                    .enable_scrolling(false)
409                    .scroll_bar_visibility(ScrollBarVisibility::AlwaysHidden)
410                    .show(ui, |ui| {
411                        ScrollArea::horizontal()
412                            .show(ui, |ui| {
413
414                                // The actual memory View Table
415                                TableBuilder::new(ui)
416                                    .resizable(false)
417                                    .striped(true)
418                                    .vscroll(false)
419                                    .cell_layout(egui::Layout::left_to_right(egui::Align::Center))
420                                    .column(Column::auto()) // Address
421                                    .column(Column::auto().at_least(78.0)) // Value
422                                    .column(Column::exact(144.0)) // Disassembly
423                                    .column(Column::remainder())// Pointers
424                                    .header(20.0, |mut header| {
425                                        self.add_table_heading(&mut header, "Addr");
426                                        self.add_table_heading(&mut header, "Value");
427                                        self.add_table_heading(&mut header, "Disassembly");
428                                        self.add_table_heading(&mut header, "");
429                                    })
430                                    .body(|mut body| {
431                                        for off in 0..=rows_to_display {
432                                            self.add_table_row(config, sender, &mut body, self.view_cache_start + off);
433                                        }
434                                    });
435                            });
436                    });
437            });
438    }
