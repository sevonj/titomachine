5//! This module contains the CPU View Panel
6//!
7
8use std::sync::mpsc::Sender;
9use egui::{Button, RichText, TopBottomPanel, Ui};
10use egui_extras::{Column, TableBody, TableBuilder};
11use crate::config::Config;
12use crate::emulator::emu_debug::CtrlMSG;
13use crate::gui::{Radix, EmulatorPanel};
14use crate::gui::FONT_TBL;
15
16/// CPUView is the GUI panel for registers.
17pub(crate) struct CPUView {
18    pub cpu_halt: bool,
19    pub cpu_cu_pc: i32,
20    pub cpu_cu_sr: i32,
21    pub cpu_gpr_r0: i32,
22    pub cpu_gpr_r1: i32,
23    pub cpu_gpr_r2: i32,
24    pub cpu_gpr_r3: i32,
25    pub cpu_gpr_r4: i32,
26    pub cpu_gpr_r5: i32,
27    pub cpu_gpr_sp: i32,
28    pub cpu_gpr_fp: i32,
29}
30
31impl CPUView {
32    pub fn new() -> Self {
33        CPUView {
34            cpu_halt: true,
35            cpu_cu_pc: 0,
36            cpu_cu_sr: 0,
37            cpu_gpr_r0: 0,
38            cpu_gpr_r1: 0,
39            cpu_gpr_r2: 0,
40            cpu_gpr_r3: 0,
41            cpu_gpr_r4: 0,
42            cpu_gpr_r5: 0,
43            cpu_gpr_sp: 0,
44            cpu_gpr_fp: 0,
45        }
46    }
47
48    /// Table Shortcut: Program counter. PC display has one difference from GPRs: it formats the
49    /// value in the same base as memory addresses.
50    fn add_row_pc(&self, body: &mut TableBody, config: &mut Config) {
51        println!("PC: {}", self.cpu_cu_pc);
52        body.row(20.0, |mut row| {
53            row.col(|ui| {
54                let value_str = config.memview_addr_base.format_i32(self.cpu_cu_pc);
55                ui.label(RichText::new(format!("PC {value_str}")).font(FONT_TBL.clone()));
56            });
57        });
58    }
59
60    /// Table Shortcut: General Purpose Registers
61    fn add_row_gpr(&self, body: &mut TableBody, config: &mut Config, name: &str, value: i32) {
62        body.row(20.0, |mut row| {
63            row.col(|ui| {
64                let value_str = config.cpuview_regs_base.format_i32(value);
65                ui.label(RichText::new(format!("{name} {value_str}")).font(FONT_TBL.clone()));
66            });
67        });
68    }
69
70    /// Table Shorcut: Status Register
71    fn add_row_sr(&self, body: &mut TableBody) {
72        body.row(20.0, |mut row| {
73            row.col(|ui| {
74                ui.label(RichText::new("Status:").font(FONT_TBL.clone()));
75                if self.cpu_halt {
76                    ui.label(RichText::new("[HALT]").font(FONT_TBL.clone()));
77                }
78            });
79        });
80        body.row(20.0, |mut row| {
81            row.col(|ui| {
82                let value_str = format!(
83                    "{}{}{}{}{}{}{}{}{}{}{}",
84                    if self.cpu_cu_sr & (1 << 31) != 0 { "G" } else { "-" },
85                    if self.cpu_cu_sr & (1 << 30) != 0 { "E" } else { "-" },
86                    if self.cpu_cu_sr & (1 << 29) != 0 { "L" } else { "-" },
87                    if self.cpu_cu_sr & (1 << 28) != 0 { "O" } else { "-" },
88                    if self.cpu_cu_sr & (1 << 27) != 0 { "Z" } else { "-" },
89                    if self.cpu_cu_sr & (1 << 26) != 0 { "U" } else { "-" },
90                    if self.cpu_cu_sr & (1 << 25) != 0 { "M" } else { "-" },
91                    if self.cpu_cu_sr & (1 << 24) != 0 { "I" } else { "-" },
92                    if self.cpu_cu_sr & (1 << 23) != 0 { "S" } else { "-" },
93                    if self.cpu_cu_sr & (1 << 22) != 0 { "P" } else { "-" },
94                    if self.cpu_cu_sr & (1 << 21) != 0 { "D" } else { "-" },
95                );
96                ui.label(RichText::new(format!("{value_str}")).font(FONT_TBL.clone()));
97            });
98        });
99    }
100}
101
102impl EmulatorPanel for CPUView {
103    fn ui(&mut self, ui: &mut Ui, config: &mut Config, _sender: &Sender<CtrlMSG>) {
104
105        // CPUView titlebar
106        TopBottomPanel::top("cpuview_titlebar")
107            .resizable(false)
108            .show_inside(ui, |ui| {
109                ui.horizontal(|ui| {
110                    let toggle_text = if config.cpuview_visible { "⏷ CPU" } else { "⏵ CPU" };
111                    if ui.add(Button::new(toggle_text).frame(false)).clicked(){
112                        config.cpuview_visible = !config.cpuview_visible;
113                    }
114                    if !config.cpuview_visible {
115                        return;
116                    }
117                    ui.menu_button("Options", |ui| {
118                        ui.label("Display registers as");
119                        if ui.radio_value(&mut config.cpuview_regs_base, Radix::Bin, "Binary").clicked() { ui.close_menu(); };
120                        if ui.radio_value(&mut config.cpuview_regs_base, Radix::Dec, "Decimal").clicked() { ui.close_menu(); };
121                        if ui.radio_value(&mut config.cpuview_regs_base, Radix::Hex, "Hex").clicked() { ui.close_menu(); };
122                    });
123                });
124            });
125
126        if !config.cpuview_visible {
127            return;
128        }
129
130        // CPUView main panel
131        TopBottomPanel::top("cpuview_main")
132            .resizable(false)
133            .show_inside(ui, |ui| {
134                TableBuilder::new(ui)
135                    .resizable(false)
136                    .striped(true)
137                    .vscroll(false)
138                    .cell_layout(egui::Layout::left_to_right(egui::Align::Center))
139                    .column(Column::auto().at_least(106.0))
140                    .body(|mut body| {
141                        self.add_row_pc(&mut body, config);
142                        self.add_row_gpr(&mut body, config, "R0", self.cpu_gpr_r0);
143                        self.add_row_gpr(&mut body, config, "R1", self.cpu_gpr_r1);
144                        self.add_row_gpr(&mut body, config, "R2", self.cpu_gpr_r2);
145                        self.add_row_gpr(&mut body, config, "R3", self.cpu_gpr_r3);
146                        self.add_row_gpr(&mut body, config, "R4", self.cpu_gpr_r4);
147                        self.add_row_gpr(&mut body, config, "R5", self.cpu_gpr_r5);
148                        self.add_row_gpr(&mut body, config, "SP", self.cpu_gpr_sp);
149                        self.add_row_gpr(&mut body, config, "FP", self.cpu_gpr_fp);
150                        self.add_row_sr(&mut body);
151                    });
152            });
153    }
154}
