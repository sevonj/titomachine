5//! Module for about window
6//!
7
8use egui::{Context, CursorIcon, Image, include_image, OpenUrl, Pos2, Vec2, Window};
9
10const VERSION: &str = env!("CARGO_PKG_VERSION");
11
12const WINDOW_SIZE: Vec2 = Vec2 { x: 192.0, y: 256.0 };
13
14/// Display about window
15pub fn about_window(ctx: &Context, open: &mut bool) {
16    // Top left corner position for a fully centered window
17    let window_topleft = ctx.input(|i| i.screen_rect.size()) / 2.0 - WINDOW_SIZE / 2.0;
18    // Same, but a Pos2 instead of Vec2. Position is nudged up slightly.
19    let window_pos = Pos2 { x: window_topleft.x, y: window_topleft.y - 64.0 };
20
21    let window = Window::new("")
22        .open(open)
23        .collapsible(false)
24        .fixed_pos(window_pos)
25        .fixed_size(WINDOW_SIZE)
26        ;
27
28    window.show(ctx, |ui| {
29        ui.set_width(ui.available_width());
30        ui.set_height(ui.available_height());
31
32        ui.vertical_centered(|ui| {
33            ui.style_mut().spacing.item_spacing = Vec2::ZERO;
34            ui.add_space(16.0);
35
36            // Title
37            ui.add(Image::new(include_image!("../assets/titobox.png")).fit_to_original_size(1.0));
38            ui.heading("TiTo");
39            ui.small(format!("titomachine {VERSION}"));
40            ui.add_space(16.0);
41
42            // Info
43            if ui.small_button("Project repository").clicked() {
44                ui.output_mut(|o| o.open_url = Some(OpenUrl {
45                    url: env!("CARGO_PKG_REPOSITORY").into(),
46                    new_tab: false,
47                }));
48            }
49            ui.add_space(24.0);
50
51            // License
52            ui.small("© 2022-2024 sevonj");
53            ui.small("Titomachine is licensed under MPL-2.0");
54            if ui.small("https://mozilla.org/MPL/2.0/")
55                .on_hover_cursor(CursorIcon::PointingHand)
56                .clicked() {
57                ui.output_mut(|o| o.open_url = Some(OpenUrl {
58                    url: "https://mozilla.org/MPL/2.0/".into(),
59                    new_tab: true,
60                }));
61            }
62        });
63    });
64}
