28pub enum Radix {
29    Bin,
30    #[default]
31    Dec,
32    Hex,
33}
34
35impl Radix {
36    pub fn format_i32(&self, value: i32) -> String {
37        match self {
38            Radix::Bin => format!("{value:#034b}"),
39            Radix::Dec => format!("{value}"),
40            Radix::Hex => format!("{value:#010x}"),
41        }
42    }
43
44    /// Same as above, but expects usize and doesn't add leading zeros.
45    pub fn format_addr(&self, value: usize) -> String {
46        match self {
47            Radix::Bin => format!("{value:#b}"),
48            Radix::Dec => format!("{value}"),
49            Radix::Hex => format!("{value:#x}"),
50        }
51    }
52}

292    fn consume_shortcuts(&mut self, ctx: &egui::Context, ui: &mut egui::Ui) {
293        if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_DEBUG_GUI)) {
294            let debug = ui.style().debug.debug_on_hover;
295            ui.ctx().set_debug_on_hover(!debug);
296            println!("its debuggin time, {}", debug)
297        }
298        // General
299        if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_NEW)) {
300            self.file_new()
301        }
302        if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_OPEN)) {
303            self.file_open()
304        }
305        if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_SAVE)) {
306            self.file_save()
307        }
308        if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_SAVEAS)) {
309            self.file_saveas()
310        }
311        if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_GUI_EDIT)) {
312            self.guimode = GuiMode::Editor
313        }
314        if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_GUI_RUN)) {
315            self.guimode = GuiMode::Emulator
316        }
317        // Editor specific
318        if self.guimode == GuiMode::Editor {
319            if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_COMPILE)) {
320                self.file_compile()
321            }
322        }
323        // Emulator specific
324        else {
325            if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_GUI_EMUGRAPHICS)) {
326                self.config.display_visible = !self.config.display_visible
327            }
328            if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_TOGGLEPOWER)) {
329                match self.emu_running {
330                    true => {
331                        let _ = self.tx_ctrl.send(CtrlMSG::PlaybackStop);
332                    }
333                    false => {
334                        let _ = self.tx_ctrl.send(CtrlMSG::PlaybackStart);
335                    }
336                }
337            }
338            if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_STOP)) {
339                let _ = self.tx_ctrl.send(CtrlMSG::PlaybackStop);
340            }
341            if self.emu_running {
342                if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_PLAY)) {
343                    let _ = self.tx_ctrl
344                        .send(CtrlMSG::PlaybackPlayPause(!self.emu_playing));
345                }
346                if ui.input_mut(|i| i.consume_shortcut(&SHORTCUT_TICK)) && !self.emu_playing {
347                    let _ = self.tx_ctrl.send(CtrlMSG::PlaybackTick);
348                    ctx.request_repaint_after(std::time::Duration::from_secs(1 / 60))
349                }
350            }
351        }

353    pub fn emulator_panel(&mut self, ctx: &Context, _: &mut Ui) {
354        // BSOD
355        if self.emu_thread_crashed {
356            EmuBSODView::new().show(ctx);
357            return;
358        }
359
360        // Refresh cached regs and memory
361        let _ = self.tx_ctrl.send(CtrlMSG::GetState);
362        let _ = self.tx_ctrl.send(CtrlMSG::GetMem(self.memoryview.get_view_cache_range()));
363
364        CentralPanel::default().show(ctx, |_| {
365            // Status Panel
366            egui::SidePanel::right("register_panel")
367                .frame(Frame::none())
368                .resizable(false)
369                // Limit max width when base isn't binary, because auto shrink doesn't work properly
370                // when separators and some other things are present.
371                .max_width(if self.config.cpuview_regs_base == Radix::Bin { 500.0 } else { 120.0 })
372                .show(ctx, |ui| {
373                    ui.set_enabled(self.is_gui_enabled());
374                    TopBottomPanel::top("status")
375                        .resizable(false)
376                        .show_inside(ui, |ui| {
377                            ui.label("Achieved speed:");
378                            ui.label(format_with_decimals_in_range(self.emu_achieved_speed as f64, 1..=1) + "%");
379                        });
380                    self.cpuview.ui(ui, &mut self.config, &self.tx_ctrl);
381                });
382
383            // IO Panel
384            egui::SidePanel::right("io_panel")
385                .frame(Frame::none())
386                .resizable(false)
387                .max_width(128.0)
388                .show(ctx, |ui| {
389                    ui.set_enabled(self.is_gui_enabled());
390                    self.legacytermview.ui(ui, &mut self.config, &self.tx_ctrl);
391                });
392
393            // Main Panel
394            CentralPanel::default()
395                .frame(Frame::none())
396                .show(ctx, |ui| {
397                    ui.set_enabled(self.is_gui_enabled());
398                    self.graphicsview.ui(ui, &mut self.config, &self.tx_ctrl);
399                    self.memoryview.ui(ui, &mut self.config, &self.tx_ctrl);
400                });
401        });
402    }
